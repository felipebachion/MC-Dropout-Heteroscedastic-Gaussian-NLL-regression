"""
General MC Dropout + Heteroscedastic Gaussian NLL regression
=============================================================

A reusable implementation for regression problems such as soil spectroscopy.

Framework
---------
This implementation follows the MC Dropout / heteroscedastic likelihood
framework described by Kendall & Gal (2017):

"""
import numpy as np
import tensorflow as tf

from tensorflow.keras import Model
from tensorflow.keras.layers import Input, Dense, Dropout, Concatenate
from tensorflow.keras.optimizers import Adam
from sklearn.model_selection import KFold
from sklearn.metrics import mean_squared_error


# ============================================================
# 1. Heteroscedastic Gaussian negative log-likelihood
# L=1/2 ​log σ^2 + ((y−μ)^2)/(2σ^2​)
# ============================================================

def heteroscedastic_loss(y_true, y_pred):

    mean = y_pred[:, 0]
    log_variance = y_pred[:, 1]

    precision = tf.exp(-log_variance)

    loss = (
        0.5 * log_variance
        + 0.5 * precision * tf.square(y_true - mean)
    )

    return tf.reduce_mean(loss)


# ============================================================
# 2. Build MC Dropout neural network
# ============================================================

def build_model(input_dim, dropout_rate=0.10, learning_rate=0.001):

    inputs = Input(shape=(input_dim,))

    x = Dense(128, activation="relu")(inputs)

    # Dropout remains active during prediction
    x = Dropout(dropout_rate)(x, training=True)

    x = Dense(64, activation="relu")(x)

    # Dropout remains active during prediction
    x = Dropout(dropout_rate)(x, training=True)

    mean = Dense(1)(x)
    log_variance = Dense(1)(x)

    outputs = Concatenate()([mean, log_variance])

    model = Model(inputs, outputs)

    model.compile(
        optimizer=Adam(learning_rate=learning_rate),
        loss=heteroscedastic_loss
    )

    return model


# ============================================================
# 3. Hyperparameter optimisation
# ============================================================

def optimise_hyperparameters(X, y):

    dropout_rates = [0.05, 0.10, 0.20]
    learning_rates = [0.0005, 0.001]
    epochs_list = [50, 100, 150]
    batch_sizes = [64, 128] # i.e. 16, 32, 64, etc.

    kfold = KFold(
        n_splits=5,
        shuffle=True,
        random_state=42
    )

    best_rmse = np.inf
    best_parameters = None

    for dropout_rate in dropout_rates:
        for learning_rate in learning_rates:
            for epochs in epochs_list:
                for batch_size in batch_sizes:

                    fold_rmse = []

                    for train_idx, val_idx in kfold.split(X):

                        X_train = X[train_idx]
                        X_val = X[val_idx]

                        y_train = y[train_idx]
                        y_val = y[val_idx]

                        model = build_model(
                            input_dim=X.shape[1],
                            dropout_rate=dropout_rate,
                            learning_rate=learning_rate
                        )

                        model.fit(
                            X_train,
                            y_train,
                            epochs=epochs,
                            batch_size=batch_size,
                            verbose=0
                        )

                        # Deterministic prediction for model selection
                        pred = model(X_val, training=False).numpy()[:, 0]

                        rmse = np.sqrt(
                            mean_squared_error(y_val, pred)
                        )

                        fold_rmse.append(rmse)

                    mean_rmse = np.mean(fold_rmse)

                    print(
                        f"Dropout={dropout_rate}, "
                        f"LR={learning_rate}, "
                        f"Epochs={epochs}, "
                        f"Batch={batch_size}, "
                        f"CV RMSE={mean_rmse:.4f}"
                    )

                    if mean_rmse < best_rmse:

                        best_rmse = mean_rmse

                        best_parameters = {
                            "dropout_rate": dropout_rate,
                            "learning_rate": learning_rate,
                            "epochs": epochs,
                            "batch_size": batch_size
                        }

    print("\nBest hyperparameters:")
    print(best_parameters)
    print(f"Best CV RMSE: {best_rmse:.4f}")

    return best_parameters


# ============================================================
# 4. MC Dropout prediction and uncertainty decomposition
# ============================================================

def predict_mc(model, X, n_mc=50):

    predictions = []

    for _ in range(n_mc):

        pred = model(X, training=True).numpy()

        predictions.append(pred)

    predictions = np.array(predictions)

    # Predicted means from all MC simulations
    means = predictions[:, :, 0]

    # Predicted conditional variances
    log_variances = predictions[:, :, 1]
    conditional_variances = np.exp(log_variances)

    # Mean prediction
    prediction = np.mean(means, axis=0)

    # Epistemic uncertainty
    epistemic_variance = np.var(means, axis=0)
    epistemic = np.sqrt(epistemic_variance)

    # Aleatoric uncertainty
    aleatoric_variance = np.mean(
        conditional_variances,
        axis=0
    )
    aleatoric = np.sqrt(aleatoric_variance)

    # Predictive uncertainty
    predictive_variance = (
        epistemic_variance
        + aleatoric_variance
    )

    predictive = np.sqrt(predictive_variance)

    return prediction, epistemic, aleatoric, predictive
